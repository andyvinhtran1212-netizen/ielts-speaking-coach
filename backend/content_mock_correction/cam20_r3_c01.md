Snapshot query (run only through an authorized read-only connection):

```sql
BEGIN READ ONLY;
SELECT to_jsonb(q) AS question,
       jsonb_build_object('id',t.id,'test_id',t.test_id,'status',t.status,
         'is_public',t.is_public,'exam_only',t.exam_only,
         'mock_content_revision',t.mock_content_revision,
         'policy_revision',t.policy_revision) AS paper
FROM public.reading_questions q
JOIN public.reading_passages p ON p.id=q.passage_id
JOIN public.reading_tests t ON t.id=p.test_id
WHERE t.test_id='MOCK0016-PROD-20261006-ILR-RDG-CAM-B20-T3-REV02'
  AND q.q_num BETWEEN 37 AND 40
ORDER BY q.q_num;
ROLLBACK;
```
